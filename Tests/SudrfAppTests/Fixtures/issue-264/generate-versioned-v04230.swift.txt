import Foundation
import XCTest
import SwiftData
import SudrfKit
@testable import SudrfApp

final class Issue264FixtureGeneratorTests: XCTestCase {
    @MainActor
    func testGenerateVersionedV1V2V3Stores() throws {
        let root = try outputRoot()
        try generate(schema: Schema(versionedSchema: SudrfSchemaV1.self),
                     name: "v1-0.42.30.store", root: root,
                     includeActs: false, includeSummaries: false)
        try generate(schema: Schema(versionedSchema: SudrfSchemaV2.self),
                     name: "v2-0.42.30.store", root: root,
                     includeActs: true, includeSummaries: false)
        try generate(schema: Schema(versionedSchema: SudrfSchemaV3.self),
                     name: "v3-0.42.30.store", root: root,
                     includeActs: true, includeSummaries: true)
    }

    @MainActor
    private func generate(schema: Schema, name: String, root: URL,
                          includeActs: Bool, includeSummaries: Bool) throws {
        let storeURL = root.appendingPathComponent(name)
        let configuration = ModelConfiguration(
            "Issue264Fixture", schema: schema, url: storeURL, cloudKitDatabase: .none)

        do {
            let container = try ModelContainer(for: schema, configurations: configuration)
            let context = ModelContext(container)
            for index in 1...2 {
                let number = "2-\(index)/2024"
                let movementContext = fixtureContext(number: number, index: index)
                let act = CaseAct(
                    id: "fixture-act-\(index)", title: "Решение",
                    date: "01.02.2024", courtShort: "Fixture court",
                    instanceLevel: .first)
                let movement = CaseMovement(
                    uid: "fixture-uid-\(index)", caseNumber: number, inForce: false,
                    instances: [], complaints: [:], acts: [act],
                    actBodies: [act.id: "Synthetic judgment text \(index)"])
                let record = TrackedCaseRecord(
                    key: movementContext.key, collections: ["Fixture", "Archive"],
                    caseNumber: number, courtTitle: movementContext.courtTitle,
                    displayDomain: movementContext.displayDomain,
                    contextData: try JSONEncoder().encode(movementContext),
                    snapshotData: try JSONEncoder().encode(fixtureSnapshot(index: index)))
                record.folderName = "Legacy folder \(index)"
                record.judicialUID = "fixture-uid-\(index)"
                record.movementData = try JSONEncoder().encode(movement)
                record.movementFetchedAt = Date(timeIntervalSince1970: 1_700_000_000 + Double(index))
                context.insert(record)

                if includeActs {
                    let document = ActDocument(
                        caseKey: movementContext.key, sourceActID: act.id,
                        caseNumber: number, judicialUID: record.judicialUID,
                        court: "Synthetic Court", instanceLevel: .first,
                        kind: act.title, date: act.date,
                        sourceText: "Synthetic judgment text \(index)")
                    let actRecord = CourtActRecord(
                        document: document, semanticKey: "fixture-\(index)",
                        fetchedAt: Date(timeIntervalSince1970: 1_700_000_100 + Double(index)))
                    context.insert(actRecord)
                    if includeSummaries {
                        context.insert(try ActSummaryRecord(
                            documentID: document.id, summary: ActSummary(),
                            provider: "fixture", model: "fixture",
                            promptVersion: "fixture-v1", pipelineVersion: "fixture-v1",
                            sourceHash: document.sourceHash,
                            generatedAt: Date(timeIntervalSince1970: 1_700_000_200 + Double(index))))
                    }
                }
            }
            try context.save()
        }
        XCTAssertTrue(FileManager.default.fileExists(atPath: storeURL.path), name)
    }

    private func outputRoot() throws -> URL {
        guard let path = ProcessInfo.processInfo.environment["ISSUE264_FIXTURE_OUTPUT"] else {
            throw XCTSkip("Set ISSUE264_FIXTURE_OUTPUT to an isolated directory.")
        }
        let root = URL(fileURLWithPath: path, isDirectory: true)
        try FileManager.default.createDirectory(at: root, withIntermediateDirectories: true)
        return root
    }

    private func fixtureContext(number: String, index: Int) -> MovementContext {
        MovementContext(
            branchRaw: CourtBranch.general.rawValue, region: "Fixture",
            searchDomain: "fixture-\(index).invalid", displayDomain: "fixture.invalid",
            courtTitle: "Synthetic Court", courtLevelRaw: CourtLevel.district.rawValue,
            courtCode: nil, cartotekaId: "g1",
            cartotekaLevelRaw: CourtLevel.district.rawValue,
            caseNumber: number, caseID: "fixture-card-\(index)",
            caseUID: "fixture-link-\(index)")
    }

    private func fixtureSnapshot(index: Int) -> CaseSnapshot {
        CaseSnapshot(
            uid: "fixture-uid-\(index)", inForce: false, category: "Synthetic",
            partiesShort: "Synthetic parties \(index)", leadCharges: nil,
            secondPartyLine: nil, stageRaw: CaseStageKind.first.rawValue,
            stageTag: "fixture", statusText: "Synthetic status",
            statusChipRaw: Palette.Chip.blue.rawValue, lastEvent: "Synthetic event",
            nextEvent: "Synthetic next event", nextChipRaw: Palette.Chip.gray.rawValue,
            steps: ["active"], sessions: [], deadlines: [], actsFingerprint: nil)
    }
}
